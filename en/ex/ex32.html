ex32: Set default order of color/symbol


title "Set default order of color/symbol"

order color default  ;# abbr. is order c
order symbol default ;# abbr. is order s

set x = range(10,1,10)

box 0 11 0 4 (gd:1 xt:1 yt:0.5)

do n 1 8
   @ exp = "log([2*n]+x)"
   set y = [exp]
   plot x y (rp:1 lg:"y=[exp]")
end

legend show r,b
